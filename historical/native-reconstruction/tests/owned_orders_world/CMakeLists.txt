cmake_minimum_required(VERSION 3.20)
project(owned_orders_world_checks LANGUAGES CXX)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(RA2_BUILD_SDL OFF CACHE BOOL "Build SDL" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Build legacy tests" FORCE)
add_subdirectory(../.. engine EXCLUDE_FROM_ALL)
enable_testing()
add_executable(owned-orders-world-vectors ../owned_orders_world_vectors.cpp)
target_link_libraries(owned-orders-world-vectors PRIVATE ra2_scenarios OpenSSL::Crypto)
add_executable(owned-orders-world-check ../owned_orders_world_check.cpp)
target_link_libraries(owned-orders-world-check PRIVATE ra2_scenarios)
foreach(target owned-orders-world-vectors owned-orders-world-check)
  target_compile_features(${target} PRIVATE cxx_std_20)
  if(RA2_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
add_test(NAME owned-orders-world-commands COMMAND owned-orders-world-check)
