cmake_minimum_required(VERSION 3.20)
project(cell_scatter_checks LANGUAGES CXX)
option(RA2_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_executable(cell-scatter-vectors ../cell_scatter_vectors.cpp)
target_include_directories(cell-scatter-vectors PRIVATE ../../src)
target_compile_features(cell-scatter-vectors PRIVATE cxx_std_20)
set_target_properties(cell-scatter-vectors PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(cell-scatter-vectors PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -frounding-math)
  if(RA2_SANITIZE)
    target_compile_options(cell-scatter-vectors PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cell-scatter-vectors PRIVATE -fsanitize=address,undefined)
  endif()
elseif(MSVC)
  target_compile_options(cell-scatter-vectors PRIVATE /W4 /fp:strict)
endif()
