cmake_minimum_required(VERSION 3.20)
project(owned_scheduled_preview_checks LANGUAGES CXX)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(RA2_BUILD_SDL OFF CACHE BOOL "Build SDL" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Build legacy tests" FORCE)
add_subdirectory(../.. engine EXCLUDE_FROM_ALL)
enable_testing()
add_executable(owned-scheduled-preview-vectors ../owned_scheduled_preview_vectors.cpp)
target_link_libraries(owned-scheduled-preview-vectors PRIVATE ra2_scenarios OpenSSL::Crypto)
add_executable(owned-scheduled-preview-check ../owned_scheduled_preview_check.cpp)
target_link_libraries(owned-scheduled-preview-check PRIVATE ra2_scenarios)
foreach(target owned-scheduled-preview-vectors owned-scheduled-preview-check)
  target_compile_features(${target} PRIVATE cxx_std_20)
  if(RA2_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
add_test(NAME owned-scheduled-preview-commands COMMAND owned-scheduled-preview-check)
