cmake_minimum_required(VERSION 3.20)
project(advance_terminal_checks LANGUAGES CXX)
option(RA2_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_executable(advance-terminal-vectors ../advance_terminal_vectors.cpp
 ../../src/sim/FootCoordinates.cpp ../../src/assets/MapFloorHeight.cpp ../../src/sim/UnitDiscovery.cpp)
target_include_directories(advance-terminal-vectors PRIVATE ../../src)
target_compile_features(advance-terminal-vectors PRIVATE cxx_std_20)
set_target_properties(advance-terminal-vectors PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
 target_compile_options(advance-terminal-vectors PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -frounding-math)
 if(RA2_SANITIZE)
  target_compile_options(advance-terminal-vectors PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(advance-terminal-vectors PRIVATE -fsanitize=address,undefined)
 endif()
elseif(MSVC)
 target_compile_options(advance-terminal-vectors PRIVATE /W4 /fp:strict)
endif()
enable_testing()
add_test(NAME advance-terminal-guards COMMAND advance-terminal-vectors --guards)
