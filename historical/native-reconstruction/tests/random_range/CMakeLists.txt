cmake_minimum_required(VERSION 3.20)
project(random_range_checks LANGUAGES CXX)
option(RA2_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_executable(random-range-vectors ../random_range_vectors.cpp ../../src/sim/OriginalRandom.cpp)
target_include_directories(random-range-vectors PRIVATE ../../src)
target_compile_features(random-range-vectors PRIVATE cxx_std_20)
set_target_properties(random-range-vectors PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
 target_compile_options(random-range-vectors PRIVATE -Wall -Wextra -Wpedantic)
 if(RA2_SANITIZE)
  target_compile_options(random-range-vectors PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(random-range-vectors PRIVATE -fsanitize=address,undefined)
 endif()
elseif(MSVC)
 target_compile_options(random-range-vectors PRIVATE /W4)
endif()
