cmake_minimum_required(VERSION 3.20)
project(nearby_location_checks LANGUAGES CXX)
option(RA2_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(nearby-location-vectors ../nearby_location_vectors.cpp
 ../../src/sim/NearbyLocation.cpp ../../src/render/Lighting.cpp
 ../../src/assets/RenderData.cpp ../../src/assets/PreparedData.cpp)
target_include_directories(nearby-location-vectors PRIVATE ../../src)
target_link_libraries(nearby-location-vectors PRIVATE OpenSSL::Crypto)
target_compile_features(nearby-location-vectors PRIVATE cxx_std_20)
set_target_properties(nearby-location-vectors PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
 target_compile_options(nearby-location-vectors PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -frounding-math)
 if(RA2_SANITIZE)
  target_compile_options(nearby-location-vectors PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(nearby-location-vectors PRIVATE -fsanitize=address,undefined)
 endif()
elseif(MSVC)
 target_compile_options(nearby-location-vectors PRIVATE /W4 /fp:strict)
endif()
enable_testing()
add_test(NAME nearby-location-guards COMMAND nearby-location-vectors --guards)
