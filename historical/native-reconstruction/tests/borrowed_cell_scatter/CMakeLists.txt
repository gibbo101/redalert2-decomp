cmake_minimum_required(VERSION 3.20)
project(borrowed_cell_scatter_checks LANGUAGES CXX)
set(RA2_BUILD_SDL OFF CACHE BOOL "Build SDL" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Build legacy tests" FORCE)
add_subdirectory(../.. engine EXCLUDE_FROM_ALL)
enable_testing()
add_executable(borrowed-cell-scatter-vectors ../borrowed_cell_scatter_vectors.cpp)
target_link_libraries(borrowed-cell-scatter-vectors PRIVATE ra2_render)
target_compile_features(borrowed-cell-scatter-vectors PRIVATE cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(borrowed-cell-scatter-vectors PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -frounding-math)
  if(RA2_SANITIZE)
    target_compile_options(borrowed-cell-scatter-vectors PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(borrowed-cell-scatter-vectors PRIVATE -fsanitize=address,undefined)
  endif()
endif()
add_executable(borrowed-cell-scatter-check ../borrowed_cell_scatter_check.cpp)
target_link_libraries(borrowed-cell-scatter-check PRIVATE ra2_render)
target_compile_features(borrowed-cell-scatter-check PRIVATE cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
 target_compile_options(borrowed-cell-scatter-check PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -frounding-math)
 if(RA2_SANITIZE)
  target_compile_options(borrowed-cell-scatter-check PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(borrowed-cell-scatter-check PRIVATE -fsanitize=address,undefined)
 endif()
endif()
add_test(NAME borrowed-cell-scatter-ownership COMMAND borrowed-cell-scatter-check)
