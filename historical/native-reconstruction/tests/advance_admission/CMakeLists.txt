cmake_minimum_required(VERSION 3.20)
project(advance_admission_checks LANGUAGES CXX)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(RA2_BUILD_SDL OFF CACHE BOOL "Build SDL" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Build legacy tests" FORCE)
add_subdirectory(../.. engine EXCLUDE_FROM_ALL)
enable_testing()
add_executable(advance-admission-vectors ../advance_admission_vectors.cpp)
target_link_libraries(advance-admission-vectors PRIVATE ra2_assets ra2_render OpenSSL::Crypto)
target_compile_features(advance-admission-vectors PRIVATE cxx_std_20)
if(RA2_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(advance-admission-vectors PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(advance-admission-vectors PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME advance-admission-guards COMMAND advance-admission-vectors --guards)
