cmake_minimum_required(VERSION 3.20)
project(owned_terrain_runtime_checks LANGUAGES CXX)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(RA2_BUILD_SDL OFF CACHE BOOL "Build SDL" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Build legacy tests" FORCE)
add_subdirectory(../.. engine EXCLUDE_FROM_ALL)
enable_testing()
add_executable(owned-terrain-runtime-check ../owned_terrain_runtime_check.cpp)
target_link_libraries(owned-terrain-runtime-check PRIVATE ra2_scenarios)
add_executable(owned-terrain-runtime-vectors ../owned_terrain_runtime_vectors.cpp)
target_link_libraries(owned-terrain-runtime-vectors PRIVATE ra2_scenarios OpenSSL::Crypto)
add_executable(owned-scatter-runtime-check ../owned_scatter_runtime_check.cpp)
target_link_libraries(owned-scatter-runtime-check PRIVATE ra2_scenarios)
foreach(target owned-terrain-runtime-check owned-terrain-runtime-vectors owned-scatter-runtime-check)
  target_compile_features(${target} PRIVATE cxx_std_20)
  if(RA2_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
add_test(NAME owned-terrain-runtime COMMAND owned-terrain-runtime-check ${CMAKE_CURRENT_SOURCE_DIR}/../../private/official-1006-probe)
set_tests_properties(owned-terrain-runtime PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_test(NAME owned-scatter-runtime COMMAND owned-scatter-runtime-check ${CMAKE_CURRENT_SOURCE_DIR}/../../private/official-1006-probe)
set_tests_properties(owned-scatter-runtime PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/../..)
