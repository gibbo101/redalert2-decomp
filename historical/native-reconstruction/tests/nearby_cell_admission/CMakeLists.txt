cmake_minimum_required(VERSION 3.20)
project(nearby_cell_admission_checks LANGUAGES CXX)
option(RA2_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_executable(nearby-cell-admission-vectors ../nearby_cell_admission_vectors.cpp
 ../../src/assets/MapCellChildren.cpp)
target_include_directories(nearby-cell-admission-vectors PRIVATE ../../src)
target_compile_features(nearby-cell-admission-vectors PRIVATE cxx_std_20)
set_target_properties(nearby-cell-admission-vectors PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
 target_compile_options(nearby-cell-admission-vectors PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -frounding-math)
 if(RA2_SANITIZE)
  target_compile_options(nearby-cell-admission-vectors PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(nearby-cell-admission-vectors PRIVATE -fsanitize=address,undefined)
 endif()
elseif(MSVC)
 target_compile_options(nearby-cell-admission-vectors PRIVATE /W4 /fp:strict)
endif()
enable_testing()
add_test(NAME nearby-cell-admission-guards COMMAND nearby-cell-admission-vectors --guards)
