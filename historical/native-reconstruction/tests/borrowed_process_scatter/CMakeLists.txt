cmake_minimum_required(VERSION 3.20)
project(borrowed_process_scatter_checks LANGUAGES CXX)
set(RA2_BUILD_SDL OFF CACHE BOOL "Build SDL" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Build legacy tests" FORCE)
add_subdirectory(../.. engine EXCLUDE_FROM_ALL)
enable_testing()
add_executable(borrowed-process-scatter-vectors ../process_terminal_scatter_vectors.cpp)
target_compile_definitions(borrowed-process-scatter-vectors PRIVATE RA2_BORROWED_PROCESS=1)
target_link_libraries(borrowed-process-scatter-vectors PRIVATE ra2_render)
target_compile_features(borrowed-process-scatter-vectors PRIVATE cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
 target_compile_options(borrowed-process-scatter-vectors PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -frounding-math)
 if(RA2_SANITIZE)
  target_compile_options(borrowed-process-scatter-vectors PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(borrowed-process-scatter-vectors PRIVATE -fsanitize=address,undefined)
 endif()
endif()
add_test(NAME borrowed-process-scatter-legacy-guards COMMAND borrowed-process-scatter-vectors --guards)

add_executable(borrowed-process-scatter-check ../borrowed_process_scatter_check.cpp)
target_link_libraries(borrowed-process-scatter-check PRIVATE ra2_render)
target_compile_features(borrowed-process-scatter-check PRIVATE cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
 target_compile_options(borrowed-process-scatter-check PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -frounding-math)
 if(RA2_SANITIZE)
  target_compile_options(borrowed-process-scatter-check PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(borrowed-process-scatter-check PRIVATE -fsanitize=address,undefined)
 endif()
endif()
add_test(NAME borrowed-process-scatter-ownership COMMAND borrowed-process-scatter-check)
