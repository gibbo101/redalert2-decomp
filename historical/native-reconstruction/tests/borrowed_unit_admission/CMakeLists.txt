cmake_minimum_required(VERSION 3.20)
project(borrowed_unit_admission_checks LANGUAGES CXX)
set(RA2_BUILD_SDL OFF CACHE BOOL "Build SDL" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Build legacy tests" FORCE)
add_subdirectory(../.. engine EXCLUDE_FROM_ALL)
enable_testing()
add_executable(borrowed-unit-admission-vectors ../borrowed_unit_admission_vectors.cpp)
add_executable(borrowed-unit-admission-check ../borrowed_unit_admission_check.cpp)
foreach(target borrowed-unit-admission-vectors borrowed-unit-admission-check)
  target_link_libraries(${target} PRIVATE ra2_render)
  target_compile_features(${target} PRIVATE cxx_std_20)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -frounding-math)
    if(RA2_SANITIZE)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
add_test(NAME borrowed-unit-admission-ownership COMMAND borrowed-unit-admission-check)
