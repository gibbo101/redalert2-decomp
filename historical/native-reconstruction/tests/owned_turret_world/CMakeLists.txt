cmake_minimum_required(VERSION 3.20)
project(owned_turret_world_checks LANGUAGES CXX)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(RA2_BUILD_SDL OFF CACHE BOOL "Build SDL" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Build legacy tests" FORCE)
add_subdirectory(../.. engine EXCLUDE_FROM_ALL)
add_executable(owned-turret-world-vectors ../owned_turret_world_vectors.cpp)
target_link_libraries(owned-turret-world-vectors PRIVATE ra2_scenarios OpenSSL::Crypto)
target_compile_features(owned-turret-world-vectors PRIVATE cxx_std_20)
if(RA2_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(owned-turret-world-vectors PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(owned-turret-world-vectors PRIVATE -fsanitize=address,undefined)
endif()
